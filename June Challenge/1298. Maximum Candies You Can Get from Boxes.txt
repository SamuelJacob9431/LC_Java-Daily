1298. Maximum Candies You Can Get from Boxes
You have n boxes labeled from 0 to n - 1. You are given four arrays: status, candies, keys, and containedBoxes where:

status[i] is 1 if the ith box is open and 0 if the ith box is closed,
candies[i] is the number of candies in the ith box,
keys[i] is a list of the labels of the boxes you can open after opening the ith box.
containedBoxes[i] is a list of the boxes you found inside the ith box.
You are given an integer array initialBoxes that contains the labels of the boxes you initially have. You can take all the candies in any open box and you can use the keys in it to open new boxes and you also can use the boxes you find in it.

Return the maximum number of candies you can get following the rules above.


Used BFS according to hint

Solution:
class Solution {
    public int maxCandies(int[] status, int[] candies, int[][] keys, int[][] containedBoxes, int[] initialBoxes) {
       int n = status.length;
       boolean[] hashKey = new boolean[n];
       boolean[] visit = new boolean[n];
       boolean[] boxAvail = new boolean[n];

       Queue<Integer> queue = new LinkedList<>();
       int totalCandies = 0;

       for(int box : initialBoxes){
        boxAvail[box] = true;
        if(status[box] == 1){
            queue.offer(box);
            visit[box] = true;
        }
       } 

       //BFS
       while(!queue.isEmpty()){
        int current = queue.poll();
        totalCandies += candies[current];

        for(int key : keys[current]){
            hashKey[key] = true;

            if(boxAvail[key] && !visit[key] && status[key] == 0){
                status[key] = 1;
                queue.offer(key);
                visit[key] = true;
            }
        }
      for (int box : containedBoxes[current]) {
                boxAvail[box] = true;   // We now have this box

                // If it is already open or we now have the key, open it
                if (!visit[box]) {
                    if (status[box] == 1 || hashKey[box]) {
                        queue.offer(box);
                        visit[box] = true;
                    }
                }
            }
        }

        return totalCandies;  // Return total candies collected
    }
}
