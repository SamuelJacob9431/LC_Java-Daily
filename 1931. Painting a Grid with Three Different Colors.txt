You are given a grid of size m x n.
You have 3 colors: let's call them Red (R), Green (G), Blue (B).

Your job is to paint the grid such that:

No two cells next to each other in a row have the same color.

No two cells above/below in a column (adjacent rows) have the same color.

Checking this properly with the help of transitions, especially considering we have 6 Valid combinations, each to be checked:
Column A: R G  (Pattern 0)
Column B: G R  (Pattern 2)

Check:
A[0] = R ≠ G = B[0] ✅
A[1] = G ≠ R = B[1] ✅
✅ Valid transition!

Solution:

import java.util.*;
class Solution {
    static final int MOD = 1_000_000_007; // basically 10^9 + 7 Prime number to prevent overflow of numbers
    public int colorTheGrid(int m, int n) {
        List<int[]> vR = new ArrayList<>(); //List to store valid row color patterns
        generatevR(m,0,new int[m], vR); // recursive function to generate valid row patterns.
        
        int len = vR.size(); //total number of valid row patterns
        Map<Integer, List<Integer>> compatible = new HashMap<>(); //Hash Map to check & tell if the pattern is allowed to prevent adjacent column wise conflict.

        for(int i = 0; i<len; i++){
            compatible .put(i,new ArrayList<>());
            for (int j = 0; j < len; j++){
                if(isCompatible(vR.get(i), vR.get(j))){
                    compatible.get(i).add(j);
                }
            }
        }

        long[] dp = new long[len];
        Arrays.fill(dp,1);

        for(int col = 1; col < n; col++){
            long[] newDp = new long[len];
            for(int i = 0; i < len; i++){
                for(int j : compatible.get(i)){
                    newDp[i] = (newDp[i] + dp[j] % MOD);
                }
            }
            dp = newDp;
        }
        long result = 0;
        for (long count : dp){
            result = (result + count) % MOD;
        }
        return (int) result;
    }

    //Generation of all valid Rows coloring recursively
    private void generatevR(int m, int pos, int[] row, List<int[]> vR){
        if(pos == m){
            vR.add(row.clone());
            return;
        } 
        for(int color = 0; color < 3; color++){
            if(pos > 0 && row[pos - 1] == color)continue; //confirms no adjacent same color
            row[pos] = color;
            generatevR(m,pos + 1, row, vR);
        }
    }

private boolean isCompatible(int[] row1, int[] row2) {
    for (int i = 0; i < row1.length; i++) {
        if (row1[i] == row2[i]) return false;
    }
    return true;
}


}
